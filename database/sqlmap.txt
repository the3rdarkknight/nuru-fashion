-- Nuru Fashion — Database Schema
-- Import this file into MySQL to set up the store.
-- Example: mysql -u root -p < schema.sql

CREATE DATABASE IF NOT EXISTS nuru_fashion CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE nuru_fashion;

-- ---------------------------------------------------------
-- Categories
-- ---------------------------------------------------------
CREATE TABLE categories (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(100) NOT NULL UNIQUE,
    image VARCHAR(255) DEFAULT NULL,
    sort_order INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------
-- Products
-- discount_percent + discount_start + discount_end together
-- drive the "daily deals" feature. An admin can set a discount
-- that only applies for a chosen date range (even a single day),
-- and the storefront automatically shows/hides it based on today's date.
-- ---------------------------------------------------------
CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    category_id INT NOT NULL,
    name VARCHAR(150) NOT NULL,
    slug VARCHAR(150) NOT NULL UNIQUE,
    description TEXT,
    base_price DECIMAL(10,2) NOT NULL,
    discount_percent DECIMAL(5,2) DEFAULT 0,
    discount_start DATE DEFAULT NULL,
    discount_end DATE DEFAULT NULL,
    sizes VARCHAR(100) DEFAULT NULL,      -- e.g. "S,M,L,XL"
    colors VARCHAR(150) DEFAULT NULL,     -- e.g. "Red,Black,Navy"
    image VARCHAR(255) DEFAULT NULL,
    stock_status ENUM('in_stock','low_stock','out_of_stock') DEFAULT 'in_stock',
    is_featured TINYINT(1) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE CASCADE
);

-- ---------------------------------------------------------
-- Admin users (for the admin panel)
-- Default login created below: username "admin", password "nuru2026"
-- CHANGE THIS PASSWORD after first login.
-- ---------------------------------------------------------
CREATE TABLE admin_users (
    id INT AUTO_INCREMENT PRIMARY KEY,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
